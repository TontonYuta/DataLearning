\begin{baitapbox}[Hệ Thống 20 Bài Tập SQL Thực Chiến]
\begin{enumerate}
    \item \textbf{Bài 1 (DDL \& Constraints)}: Viết câu lệnh DDL tạo bảng \texttt{dim\_employees} với các ràng buộc chuẩn: \texttt{emp\_id VARCHAR(50)} là khóa chính, \texttt{email VARCHAR(100)} duy nhất, \texttt{salary NUMERIC(12, 2)} phải lớn hơn 0 (\texttt{CHECK}), và \texttt{hire\_date DATE} mặc định là ngày hiện tại.
    \item \textbf{Bài 2 (Conditional Update)}: Viết câu lệnh cập nhật tăng lương 10\% cho tất cả nhân viên thuộc phòng ban \texttt{'Engineering'} có ngày tuyển dụng trước ngày hiện tại từ 2 năm trở lên.
    \item \textbf{Bài 3 (Tìm Khách Hàng Chưa Từng Mua Hàng -- Bảng \texttt{customers} \& \texttt{orders})}: Dựa trên tệp \href{run:./data/customers.csv}{\texttt{data/customers.csv}} và \href{run:./data/orders.csv}{\texttt{data/orders.csv}}, viết truy vấn tìm tất cả khách hàng chưa từng phát sinh bất kỳ đơn hàng nào bằng 2 cách: (a) Dùng \texttt{LEFT JOIN}; (b) Dùng \texttt{NOT EXISTS}. \textit{Kết quả kiểm tra}: Có chính xác \textbf{11 khách hàng} chưa từng mua hàng.
    \item \textbf{Bài 4 (Aggregation với HAVING -- Bảng \texttt{order\_items})}: Viết truy vấn nhóm theo ngành hàng (\texttt{category}), tính tổng số mặt hàng bán ra và tổng doanh thu. Chỉ hiển thị các ngành hàng có doanh số vượt quá 1.6 triệu USD. \textit{Kết quả kiểm tra}: Cả 5 ngành hàng đều vượt ngưỡng, đứng đầu là \textbf{Fashion} với \textbf{\$1,684,668.49}.
    \item \textbf{Bài 5 (Bẫy Nhân Đôi Dữ Liệu -- Fan-Out Trap Trong JOIN)}: Một đơn hàng có thể có nhiều món hàng (\texttt{order\_items}) và được thanh toán qua nhiều cổng (\texttt{payments}). Nếu viết câu lệnh \texttt{orders JOIN order\_items JOIN payments} rồi tính \texttt{SUM(payments.payment\_value)}, tại sao tổng tiền bị sai gấp nhiều lần? Viết câu lệnh dùng CTE tiền tổng hợp để giải quyết triệt để lỗi Fan-Out.
    \item \textbf{Bài 6 (Self Join -- Cặp Đôi Cùng Thành Phố)}: Trên bảng \texttt{customers}, viết truy vấn tìm các cặp khách hàng cùng sinh sống tại thành phố \texttt{'Da Nang'}, đảm bảo không trùng lặp chính họ (\texttt{id1 <> id2}) và không trùng hoán vị (\texttt{id1 < id2}).
    \item \textbf{Bài 7 (Window Function -- Xếp Hạng Doanh Thu Ngành Hàng)}: Viết truy vấn tính tổng GMV của từng ngành hàng trong \texttt{order\_items}, sử dụng \texttt{DENSE\_RANK() OVER (ORDER BY GMV DESC)} để xếp hạng doanh thu từ cao xuống thấp.
    \item \textbf{Bài 8 (Running Total Cộng Dồn Theo Thời Gian)}: Viết truy vấn sử dụng hàm \lstinline|SUM(payment_value) OVER (ORDER BY order_purchase_timestamp ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)| để tính tổng doanh thu tích lũy cộng dồn qua từng đơn hàng.
    \item \textbf{Bài 9 (Tính Thời Gian Đơn Hàng Lưu Kho Bằng LEAD/LAG)}: Cho bảng nhật ký vận chuyển \texttt{order\_tracking(order\_id, status, update\_time)}. Viết câu lệnh dùng \texttt{LEAD(update\_time)} để tính số giờ chênh lệch giữa lúc đơn ở trạng thái \texttt{'IN\_WAREHOUSE'} và khi chuyển sang \texttt{'SHIPPING'}.
    \item \textbf{Bài 10 (Window Function -- NTILE Phân Hạng Khách Hàng)}: Dựa trên tổng chi tiêu của khách hàng, sử dụng hàm \texttt{NTILE(4) OVER (ORDER BY total\_spend DESC)} để chia khách hàng thành 4 nhóm tứ phân vị (Top 25\% VIP nhất là Quartile 1, 25\% thấp nhất là Quartile 4).
    \item \textbf{Bài 11 (Tăng Trưởng Doanh Thu MoM Bằng LAG)}: Viết truy vấn gộp nhóm doanh thu theo tháng \texttt{YYYY-MM}, dùng CTE kết hợp \texttt{LAG(monthly\_revenue, 1) OVER (ORDER BY month)} để tính tỷ lệ tăng trưởng doanh số so với tháng trước (\% MoM Growth).
    \item \textbf{Bài 12 (Tìm Chuỗi Hoạt Động Liên Tục -- Gaps \& Islands)}: Cho bảng \texttt{user\_logins(user\_id, login\_date)}. Dùng kỹ thuật chênh lệch ngày \texttt{login\_date - ROW\_NUMBER() OVER (PARTITION BY user\_id ORDER BY login\_date)} để tìm tất cả người dùng đăng nhập liên tục từ 7 ngày trở lên.
    \item \textbf{Bài 13 (Recursive CTE -- Cây Phân Cấp Danh Mục)}: Cho bảng danh mục sản phẩm lồng nhau \texttt{categories(cat\_id, name, parent\_id)}. Viết một Recursive CTE xuất ra cây phả hệ đầy đủ kèm cấp bậc độ sâu (Depth Level: 1, 2, 3...) và đường dẫn danh mục (Path: \texttt{"Electronics > Mobile > iPhone"}).
    \item \textbf{Bài 14 (Pivot Dữ Liệu Bằng CASE WHEN)}: Viết truy vấn thống kê số lượng đơn hàng theo từng thành phố và xoay ngang 3 trạng thái đơn hàng (\texttt{COMPLETED, CANCELLED, REFUNDED}) thành 3 cột riêng biệt thay vì dùng lệnh GROUP BY dạng dọc.
    \item \textbf{Bài 15 (Truy Vấn Tương Quan -- Correlated Subquery)}: Viết truy vấn tìm tất cả các mặt hàng trong \texttt{order\_items} có đơn giá (\texttt{price}) cao hơn đơn giá trung bình của chính ngành hàng (\texttt{category}) mà mặt hàng đó trực thuộc.
    \item \textbf{Bài 16 (Khử Trùng Lặp Bản Ghi Bằng CTE \& ROW\_NUMBER)}: Khi bảng staging bị ghi đè trùng 3 bản ghi giống hệt nhau về \texttt{order\_id}, viết câu lệnh \texttt{DELETE} sử dụng CTE và \texttt{ROW\_NUMBER() OVER (PARTITION BY order\_id ORDER BY loaded\_at DESC)} để chỉ giữ lại duy nhất bản ghi mới nhất.
    \item \textbf{Bài 17 (Phân Tích Khách Hàng VIP Nhất -- Bảng \texttt{customers} \& \texttt{orders})}: Dựa trên tệp \href{run:./data/orders.csv}{\texttt{data/orders.csv}} và \href{run:./data/order_items.csv}{\texttt{data/order\_items.csv}}, viết truy vấn tìm khách hàng đã chi tiêu nhiều nhất ở các đơn \texttt{'COMPLETED'}. \textit{Kết quả kiểm tra}: Khách hàng \textbf{CUST\_01815} ("Khach Hang 1815", Da Nang) dẫn đầu với \textbf{9 đơn hàng} và tổng chi tiêu \textbf{\$11,306.13}.
    \item \textbf{Bài 18 (Tối Ưu Hóa Query Chậm -- Sargable Query)}: Câu lệnh \texttt{SELECT * FROM orders WHERE DATE(order\_purchase\_timestamp) = '2026-03-01'} không thể sử dụng B-Tree Index trên cột timestamp. Hãy viết lại câu lệnh theo chuẩn Sargable để database engine thực hiện Index Range Scan thay vì Full Table Scan.
    \item \textbf{Bài 19 (Chỉ Mục Bao Phủ -- Covering Index)}: Giải thích cơ chế Covering Index. Viết câu lệnh DDL tạo chỉ mục trong PostgreSQL giúp câu lệnh sau đạt tốc độ Index Only Scan không cần chạm vào Heap Table: \texttt{SELECT order\_id, order\_status FROM orders WHERE customer\_id = 'CUST\_00001';}.
    \item \textbf{Bài 20 (Cập Nhật Bổ Sung Dữ Liệu -- UPSERT)}: Viết câu lệnh nạp bản ghi vào bảng \texttt{dim\_customer\_stats}: Nếu \texttt{customer\_id} chưa có thì chèn mới, nếu đã có thì cập nhật \texttt{total\_spend = total\_spend + EXCLUDED.new\_spend} và \texttt{last\_active = NOW()} bằng mệnh đề \texttt{ON CONFLICT (customer\_id) DO UPDATE}.
\end{enumerate}
\end{baitapbox}

\begin{dapanbox}[Đáp Án \& Lời Giải Ngắn Gọn Cho 20 Bài Tập SQL]
\begin{itemize}
    \item \textbf{Bài 1}: 
    \begin{lstlisting}[language=SQL]
CREATE TABLE dim_employees (
    emp_id VARCHAR(50) PRIMARY KEY,
    email VARCHAR(100) UNIQUE NOT NULL,
    salary NUMERIC(12, 2) CHECK (salary > 0),
    hire_date DATE DEFAULT CURRENT_DATE
);
    \end{lstlisting}
    \item \textbf{Bài 2}: \texttt{UPDATE dim\_employees SET salary = salary * 1.10 WHERE dept = 'Engineering' AND hire\_date <= CURRENT\_DATE - INTERVAL '2 years';}.
    \item \textbf{Bài 3}: Cách 1: \texttt{SELECT c.* FROM customers c LEFT JOIN orders o ON c.customer\_id = o.customer\_id WHERE o.order\_id IS NULL;}. Cách 2: \texttt{SELECT * FROM customers c WHERE NOT EXISTS (SELECT 1 FROM orders o WHERE o.customer\_id = c.customer\_id);}. Cả 2 cách đều trả về chính xác \textbf{11 khách hàng}.
    \item \textbf{Bài 4}: 
    \begin{lstlisting}[language=SQL]
SELECT category, COUNT(*) AS total_items, ROUND(SUM(price * quantity), 2) AS total_gmv
FROM order_items
GROUP BY category
HAVING SUM(price * quantity) > 1600000
ORDER BY total_gmv DESC;
    \end{lstlisting}
    Top 1: Fashion (\$1,684,668.49); Top 2: Beauty (\$1,654,187.31); Top 3: Electronics (\$1,649,725.51).
    \item \textbf{Bài 5}: Tiền tổng hợp thanh toán trước khi JOIN:
    \begin{lstlisting}[language=SQL]
WITH aggregated_payments AS (
    SELECT order_id, SUM(payment_value) AS total_paid
    FROM payments GROUP BY order_id
)
SELECT o.customer_id, ROUND(SUM(p.total_paid), 2) AS actual_spend
FROM orders o JOIN aggregated_payments p ON o.order_id = p.order_id
GROUP BY o.customer_id;
    \end{lstlisting}
    \item \textbf{Bài 6}: \texttt{SELECT c1.customer\_name, c2.customer\_name, c1.city FROM customers c1 JOIN customers c2 ON c1.city = c2.city AND c1.customer\_id < c2.customer\_id WHERE c1.city = 'Da Nang';}.
    \item \textbf{Bài 7}: \texttt{SELECT category, SUM(price*quantity) AS gmv, DENSE\_RANK() OVER (ORDER BY SUM(price*quantity) DESC) AS rank\_no FROM order\_items GROUP BY category;}.
    \item \textbf{Bài 8}: \lstinline|SUM(payment_value) OVER (ORDER BY order_purchase_timestamp ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)|.
    \item \textbf{Bài 9}: \texttt{ROUND(EXTRACT(EPOCH FROM (LEAD(update\_time) OVER(PARTITION BY order\_id ORDER BY update\_time) - update\_time)) / 3600.0, 2)}.
    \item \textbf{Bài 10}: \texttt{NTILE(4) OVER (ORDER BY total\_spend DESC)}. Phân khúc khách hàng thành 4 hạng cân bằng.
    \item \textbf{Bài 11}: \lstinline|ROUND((rev - LAG(rev, 1) OVER (ORDER BY month)) / LAG(rev, 1) OVER (ORDER BY month) * 100.0, 2)|.
    \item \textbf{Bài 12}: Gom nhóm đảo bằng \lstinline|login_date - ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) * INTERVAL '1 day' AS grp|, sau đó \texttt{GROUP BY user\_id, grp HAVING COUNT(*) >= 7}.
    \item \textbf{Bài 13}:
    \begin{lstlisting}[language=SQL]
WITH RECURSIVE cat_tree AS (
    SELECT cat_id, name, parent_id, 1 AS depth, CAST(name AS VARCHAR(500)) AS path
    FROM categories WHERE parent_id IS NULL
    UNION ALL
    SELECT c.cat_id, c.name, c.parent_id, ct.depth + 1, CAST(ct.path || ' > ' || c.name AS VARCHAR(500))
    FROM categories c JOIN cat_tree ct ON c.parent_id = ct.cat_id
) SELECT * FROM cat_tree ORDER BY depth, path;
    \end{lstlisting}
    \item \textbf{Bài 14}: 
    \begin{lstlisting}[language=SQL]
SELECT c.city,
    COUNT(CASE WHEN o.order_status = 'COMPLETED' THEN 1 END) AS completed_orders,
    COUNT(CASE WHEN o.order_status = 'CANCELLED' THEN 1 END) AS cancelled_orders,
    COUNT(CASE WHEN o.order_status = 'REFUNDED' THEN 1 END) AS refunded_orders
FROM customers c JOIN orders o ON c.customer_id = o.customer_id
GROUP BY c.city;
    \end{lstlisting}
    \item \textbf{Bài 15}: \texttt{SELECT * FROM order\_items oi WHERE price > (SELECT AVG(price) FROM order\_items sub WHERE sub.category = oi.category);}.
    \item \textbf{Bài 16}: \texttt{WITH ranked AS (SELECT ctid, ROW\_NUMBER() OVER(PARTITION BY order\_id ORDER BY loaded\_at DESC) AS rn FROM stg\_orders) DELETE FROM stg\_orders WHERE ctid IN (SELECT ctid FROM ranked WHERE rn > 1);}.
    \item \textbf{Bài 17}: Truy vấn khách VIP nhất:
    \begin{lstlisting}[language=SQL]
WITH cust_spend AS (
    SELECT c.customer_id, c.customer_name, c.city, COUNT(DISTINCT o.order_id) AS order_count,
           ROUND(SUM(oi.price * oi.quantity), 2) AS total_spend
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    JOIN order_items oi ON o.order_id = oi.order_id
    WHERE o.order_status = 'COMPLETED'
    GROUP BY c.customer_id, c.customer_name, c.city
)
SELECT * FROM cust_spend ORDER BY total_spend DESC LIMIT 1;
    \end{lstlisting}
    Kết quả: \textbf{CUST\_01815} ("Khach Hang 1815", Da Nang), 9 đơn, chi tiêu \textbf{\$11,306.13}.
    \item \textbf{Bài 18}: Viết lại dạng Sargable: \texttt{WHERE order\_purchase\_timestamp >= '2026-03-01 00:00:00' AND order\_purchase\_timestamp < '2026-03-02 00:00:00'}.
    \item \textbf{Bài 19}: \texttt{CREATE INDEX idx\_orders\_cust\_status ON orders (customer\_id) INCLUDE (order\_id, order\_status);}. Giúp query chỉ đọc index leaf pages mà không cần truy xuất bảng vật lý.
    \item \textbf{Bài 20}: 
    \begin{lstlisting}[language=SQL]
INSERT INTO dim_customer_stats (customer_id, total_spend, last_active)
VALUES ('CUST_00001', 250.00, CURRENT_TIMESTAMP)
ON CONFLICT (customer_id) DO UPDATE SET
    total_spend = dim_customer_stats.total_spend + EXCLUDED.total_spend,
    last_active = EXCLUDED.last_active;
    \end{lstlisting}
\end{itemize}
